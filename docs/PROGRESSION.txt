# admit_pending = true при КАЖДОМ включении (не сохраняется в NVS).
# Снимается чипом type=3 sub=6 «ДОПУСК В ИГРУ» (CmdSub.ADMIT) — только главный мастер.
# is_dead сохраняется в NVS → экран «СВЯЗЬ ПОТЕРЯНА».
# Снимается чипом type=3 sub=0 «ВОСКРЕШЕНИЕ» (CmdSub.REVIVE) — любой мастер поля.
# Играть можно только когда: not admit_pending AND not is_dead.

# =============================================================================
# 1. СЛОТЫ TCA9548A (TRRS) — ФИНАЛЬНАЯ СХЕМА
# =============================================================================
# Эталон в коде: proshivki/common/mux_channels.h
#
# CH0 — УНИВЕРСАЛЬНЫЙ: расходник, админка, магазин, квест, банкомат, кабель кассы
#       Handoff: заголовок чипа @0x00 → вставка картриджа.
#       Кабель: настольная кассета + 4-провод к CH0; TXN-блок @0x80 (eeprom_txn.h).
# CH1, CH4 — РЕЗЕРВ (не используется на PCB)
# CH2, CH3, CH5, CH6 — универсальные слоты снаряжения (броня / артефакт по полю type чипа)
# CH7 — запасной
#
# LoRa для кассы/квестов НЕ используется — только I2C/EEPROM через TCA.

# =============================================================================
# 2. УРОВНИ (1–100)
# =============================================================================
# Уровни растут АВТОМАТИЧЕСКИ по накопленному XP.
# При повышении: +RUB награда (НЕ списание).
# Ранг — отдельно (см. §3).
#
# Код: game_logic.xp_per_level / xp_total_for_level.
# Ребаланс v1.2 (июнь 2026): квест XP = RUB за сдачу; пороги под активную игру.

# --- Цели темпа (АКТИВНАЯ игра: квесты, аномалии, арена — не AFK в укрытии) ---
#   Ур. 1→10:  ~1–2 ч активной игры (макс. ~3 ч)
#   Ур. 1→100: ~6–7 полных игровых дней при макс. активности (~4–6 ч/день)
#   Типичная сессия ≈ 1 игровой день (~4 000–5 500 XP при 8–12 квестах)
# Оценка XP/ч: ранняя ~1 200; середина ~1 500; поздняя ~1 800 (квесты + поле + достижения)

# --- Формула XP на шаг (уровень L → L+1) — четыре зоны ---
#   L  1–9:   90 + L×32           быстрый старт (регистрация, первые квесты)
#   L 10–39:  200 + (L−9)×8       ранний/средний — квесты и аномалии
#   L 40–69:  420 + (L−39)×4      ветеранский гринд
#   L 70–99:  460 + (L−69)×3      финальный сезон
#
# Суммарный XP до уровня N = сумма шагов 1…N−1.

# Ключевые уровни (XP всего | ориентир активного времени | пример пути):
#   1:     0    | старт
#   2:   122    | ~15 мин — регистрация + 1–2 мелких достижения
#   5:   680    | ~30 мин — +2–3 аномалии, первая покупка
#  10: 2 250    | ~1.5–2 ч — 5–7 квестов (300 RUB) + 5 MAC + стартовые достижения
#  20: 4 690    | ~1 игр. день — 10+ квестов, 8–10 аномалий
#  25: 6 210    | ~1.5 дня — ранг СТАЛКЕР, детектор
#  40: 11 970   | ~3 дня — 25+ квестов, коллекция аномалий
#  50: 16 390   | ~4 дня — торговля/арена (достижения E/F/L)
#  80: 31 195   | ~5.5 дня — 50+ квестов, большинство категорий достижений
# 100: 41 625   | ~6–7 дней макс. активности — полный сезон

# Источники XP (основной — квесты):
#   Сдача квеста: XP = RUB награда чипа (каждый квест, не только вехи G01–G07)
#   Вехи G01–G07: бонус XP поверх (≈50% от старых значений)
#   Первая уникальная аномалия (MAC): +40 XP + достижение B01 (150 XP)
#   Прочие достижения: см. §7

# 10. ЕДИНОЕ ПРИЛОЖЕНИЕ STALKER (ПК)

# Статус: планирование (июнь 2026). ROADMAP.md — ФАЗА 1.5. Полный UI — после согласования фаз.
# Одна программа «STALKER» на ноутбуке мастера — главное меню + модули, не отдельные exe.
#
# Модули (целевое меню):
#   [ Регистрация ]        — привязка PDA ↔ player_id, общая БД события
#   [ Программатор ]       — programmat_pc/ (чипы, аномалии, убежища, ПДА, терминалы)
#   [ Сбор статистики ]    — съём с ПДА в конце игры/смены
#   [ Глобальное сообщение ] — позже (§11)
#   [ Карта локации ]      — позже (§11)
#
# Принципы:
#   Регистрация + Сбор статистики — один блок «база игроков события», SQLite, USB экспорт/импорт.
#   Программатор — существующий функционал programmat_pc/ без потери возможностей.
#   Связь с железом: USB Serial к ОДНОМУ устройству мастера (CHIP_BOX / пульт / телефон).
#   ПДА игрока: LoRa по № регистрации ИЛИ общий EEPROM-чип. Не кабель к каждому ПДА.
#   Регистрация на терминале (§9.3) — полигон; ПК-модуль — центральная база и массовые операции.

# --- 10.1. Модули (кратко) ---
# Регистрация     | ID игрока = адрес LoRa; UID выдаёт мастер; мост CHIP_BOX+шнур (TXN REGISTER)
# Программатор    | Tk-фрейм в stalker_app (чипы, аномалии, убежища, пресет ПДА, роли)
# Сбор статистики | Съём: уровень, XP, деньги, смерти, достижения → БД; чип «Сохранение» — позже
# Глоб. сообщение | Мастер-Пульт → LoRa по ID; UI минуты выброса; радио=трек; громкость на ПДА
# Карта           | Скриншот + маяки; номер железки появляется сам при подключении шнуром

# --- 10.2. Общая база игроков (SQLite, эскиз) ---
# Файл на событие: stalker_event_<event_id>.db (переносимый, USB).
#
# events:
#   event_id TEXT PK | title | created_at | closed_at NULL
#
# players:
#   player_id INTEGER PK (в рамках события)
#   event_id FK | name | callsign | group_name
#   pda_uid TEXT NULL (MAC/UID EEPROM — привязка железа)
#   registered_at | registered_by NULL | notes NULL
#   Уникальность: (event_id, pda_uid) где pda_uid не NULL; (event_id, player_id)
#
# player_stats (снимок в конце игры):
#   player_id FK | event_id FK | collected_at
#   level | xp | money_rub | deaths | cheat_shield_count
#   achievements_json | rank_title NULL | pda_snapshot_json NULL
#
# Экспорт: копия .db + опционально players.csv / stats.csv на USB.
# Импорт: «Открыть событие» с флешки; конфликт player_id/pda_uid — предупреждение.
# Протокол регистрации: CONFIG:REGISTER:name=... (+ позже callsign/group).
# Константы достижений — programmat_pc/game_logic.py (ACHIEVEMENTS, ранги).
